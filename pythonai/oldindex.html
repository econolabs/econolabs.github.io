🔍 FAISS — не работает, но можно написать kNN самим

Проблема: FAISS содержит оптимизированный C++ код, поэтому несовместим с Pyodide.

Учебная альтернатива: Реализовать наивный поиск ближайших соседей (kNN) через NumPy. Для небольшого количества векторов (до 1000) это работает быстро и прекрасно иллюстрирует принцип. 

def naive_knn(query_vector, vectors, k=3):
    # query_vector: эмбеддинг запроса
    # vectors: все эмбеддинги в базе
    # k: сколько ближайших вернуть
    distances = np.linalg.norm(vectors - query_vector, axis=1)
    nearest_indices = np.argsort(distances)[:k]
    return nearest_indices, distances[nearest_indices]
🎓 Педагогическая ценность: Студенты поймут, как работают ANN (приближённые методы) — сначала наивный поиск (медленно, точно), затем FAISS (быстро).

Нейросеть с нуля — один нейрон, который обучается

🧠 numpy

Реализуем простейший перцептрон и обучим его на логическом ИЛИ — поймём, как работают нейросети внутри.

# ЭТАП 1: Импорт библиотек и определение активационной функции
import numpy as np

# Активационная функция (ступенька)
# Возвращает 1, если сумма >= 0, иначе 0
def step_function(x):
    return 1 if x >= 0 else 0
# ЭТАП 2: Реализация класса Perceptron
class Perceptron:
    def __init__(self, n_inputs, learning_rate=0.1):
        # Инициализация весов и смещения случайными числами
        self.weights = np.random.randn(n_inputs)
        self.bias = np.random.randn()
        self.lr = learning_rate  # Скорость обучения
    
    def predict(self, inputs):
        # Вычисляем взвешенную сумму
        linear = np.dot(inputs, self.weights) + self.bias
        return step_function(linear)
    
    def train(self, X, y, epochs=20):
        # Обучение перцептрона
        for epoch in range(epochs):
            errors = 0
            for inputs, target in zip(X, y):
                prediction = self.predict(inputs)
                error = target - prediction
                if error != 0:
                    # Правило Хебба: корректируем веса
                    self.weights += self.lr * error * np.array(inputs)
                    self.bias += self.lr * error
                    errors += 1
            if epoch % 5 == 0:
                print(f"Эпоха {epoch}: ошибок {errors}")
# ЭТАП 3: Обучение перцептрона для задачи логического ИЛИ
# Таблица истинности для ИЛИ:
# 0 OR 0 = 0
# 0 OR 1 = 1
# 1 OR 0 = 1
# 1 OR 1 = 1

X_train = [[0,0], [0,1], [1,0], [1,1]]
y_train = [0, 1, 1, 1]

p = Perceptron(n_inputs=2, learning_rate=0.1)
p.train(X_train, y_train, epochs=20)
# ЭТАП 4: Проверка обученного перцептрона
print("\n🔍 ПРОВЕРКА ОБУЧЕННОГО ПЕРЦЕПТРОНА (ЛОГИЧЕСКОЕ ИЛИ):")
print("="*45)

for inputs in X_train:
    result = p.predict(inputs)
    print(f" {inputs[0]} OR {inputs[1]} = {result}")

# Вывод весов обученной модели
print(f"\n📊 Веса модели: {p.weights.round(3)}")
print(f"📊 Смещение (bias): {p.bias:.3f}")
# ЭТАП 5: ЗАДАНИЕ ДЛЯ СТУДЕНТОВ
print("\n" + "="*50)
print("💡 ЗАДАНИЕ: ПОПРОБУЙТЕ ОБУЧИТЬ ПЕРЦЕПТРОН НА ЛОГИЧЕСКОЕ 'И'")
print("="*50)

print("""
📝 Таблица истинности для логического 'И' (AND):
0 AND 0 = 0
0 AND 1 = 0
1 AND 0 = 0
1 AND 1 = 1

🔧 Замените y_train на:
y_train = [0, 0, 0, 1]

❓ Вопросы для исследования:

📄 5. RAG — заставляем ИИ отвечать по вашим документам

🧩 transformers + faiss

Retrieval-Augmented Generation: сначала ищем нужный фрагмент, потом генерируем ответ на его основе. 

# Карточка 5: RAG — чат с вашими документами

%pip install sentence-transformers faiss-cpu

import numpy as np
from sentence_transformers import SentenceTransformer
import faiss

# Наш учебный документ (короткая инструкция)
documents = [
    "JupyterLite позволяет запускать Python в браузере без сервера.",
    "Для установки библиотек используйте %pip install или await piplite.install().",
    "ИИ-модели могут работать прямо в браузере через WebAssembly.",
    "RAG расшифровывается как Retrieval-Augmented Generation.",
    "Сегодня мы изучаем практические инструменты искусственного интеллекта."
]

# Загружаем эмбеддер (превращает текст в числа)
model = SentenceTransformer('all-MiniLM-L6-v2')

# Создаём векторную базу
embeddings = model.encode(documents)
dimension = embeddings.shape[1]
index = faiss.IndexFlatL2(dimension)
index.add(embeddings.astype(np.float32))

# Функция поиска
def retrieve(query, k=2):
    query_embedding = model.encode([query])
    distances, indices = index.search(query_embedding.astype(np.float32), k)
    return [documents[i] for i in indices[0]]

# Пример поиска
question = "Как установить библиотеку в JupyterLite?"
relevant_chunks = retrieve(question)
print(f"Вопрос: {question}")
print("Найденные релевантные фрагменты:")
for chunk in relevant_chunks:
    print(f"  - {chunk}")

# Имитация генерации ответа (настоящий RAG использует LLM)
print("\n💡 Смоделированный ответ ИИ:")
print("Чтобы установить библиотеку в JupyterLite, используйте команду %pip install или await piplite.install().")

# Карточка 9: Нейросеть на Keras — XOR

%pip install tensorflow

import numpy as np
import tensorflow as tf
from tensorflow.keras.models import Sequential
from tensorflow.keras.layers import Dense

# Данные XOR: входы и правильные ответы
X = np.array([[0, 0], [0, 1], [1, 0], [1, 1]], dtype=np.float32)

    pred = model.predict(inputs.reshape(1,2), verbose=0)[0][0]

🎨 10. Принцип работы диффузионных моделей — симуляция шумоподавления

🖌️ Диффузия

Генерация изображений вроде Stable Diffusion работает так: модель постепенно убирает шум из случайного пятна, превращая его в осмысленную картинку. 

# Карточка 10: Принцип диффузионных моделей (симуляция)

%pip install numpy matplotlib

import numpy as np
import matplotlib.pyplot as plt

def add_noise(image, noise_level):
    """Добавляет шум к изображению"""
    noise = np.random.randn(*image.shape) * noise_level
    return np.clip(image + noise, 0, 1)

def denoise_step(noisy_image, step, total_steps):
    """Симуляция одного шага денойзинга (упрощённо)"""
    # В реальной модели это делает нейросеть
    alpha = step / total_steps
    denoised = noisy_image * (1 + alpha) - alpha * np.random.randn(*noisy_image.shape) * 0.1
    return np.clip(denoised, 0, 1)

# Создаём простое "изображение" (цифра 8 в виде матрицы)
clean_image = np.zeros((28, 28))
clean_image[10:18, 10:18] = 1  # квадрат 8x8
# Рисуем круг (упрощённо)
for i in range(28):
    for j in range(28):
        if (i-14)**2 + (j-14)**2 < 36:
            clean_image[i, j] = 1

# Добавляем шум
noisy = add_noise(clean_image, noise_level=0.5)

# Симулируем процесс денойзинга
steps = 10
current = noisy.copy()
fig, axes = plt.subplots(1, steps+2, figsize=(15, 3))
axes[0].imshow(clean_image, cmap='gray')
axes[0].set_title("Оригинал")
axes[1].imshow(noisy, cmap='gray')
axes[1].set_title("Зашумлённое")

for i in range(steps):
    current = denoise_step(current, i+1, steps)
    axes[i+2].imshow(current, cmap='gray')
    axes[i+2].set_title(f"Шаг {i+1}")

for ax in axes:
    ax.axis('off')
plt.tight_layout()
plt.show()

print("Это симуляция обратного диффузионного процесса.")
print("Реальные модели (Stable Diffusion, DALL-E) обучаются предсказывать шум, который нужно убрать.")
print("На входе — случайный шум, на выходе — осмысленное изображение.")
✨ Диффузионные модели — основа Midjourney, DALL-E и Stable Diffusion. Они превращают случайный шум в красивое изображение за сотни шагов.
